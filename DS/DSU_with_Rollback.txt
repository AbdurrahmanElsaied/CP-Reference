#include <iostream>
#include <vector>
#include <numeric>

struct DSUWithRollback {
    std::vector<int> parent, size;
    struct Operation {
        int u, v;
        int prev_size_u;
    };
    std::vector<Operation> history; 
    DSUWithRollback(int n) {
        parent.resize(n);
        std::iota(parent.begin(), parent.end(), 0);
        size.assign(n, 1);
    }
    
    int find(int u) {
        while (u != parent[u]) {
            u = parent[u];
        }
        return u;
    }

    bool unite(int u, int v) {
        int root_u = find(u);
        int root_v = find(v);
        if (root_u == root_v) return false;

        // Union by Size
        if (size[root_u] < size[root_v]) std::swap(root_u, root_v);
        history.push_back({root_u, root_v, size[root_u]});
        parent[root_v] = root_u;
        size[root_u] += size[root_v];
        return true;
    }

    void rollback() {
        if (history.empty()) return;
        Operation last_op = history.back();
        history.pop_back();


        size[last_op.u] = last_op.prev_size_u;
        parent[last_op.v] = last_op.v; // فصل الشجرة التي تم دمجها
    }

   
    int time() {
        return history.size();
    }

 
    void revert_to(int t) {
        while (history.size() > t) {
            rollback();
        }
    }
};
